            "D. Untuk membatasi jumlah penyintas yang ingin meminta bantuan.",
            "E. Karena pemeriksaan keamanan adalah formalitas administrasi sebelum membagikan obat."
        ],
        answer: 1
    }
];

let currentQuestion = 0;
let score = 0;

const questionNumberEl = document.getElementById('question-number');
const questionTextEl = document.getElementById('question-text');
const optionsContainerEl = document.getElementById('options-container');
const nextBtn = document.getElementById('next-btn');
const scoreCounterEl = document.getElementById('score-counter');
const progressBarEl = document.getElementById('progress-bar');

function loadQuestion() {
    const q = quizData[currentQuestion];
    questionNumberEl.innerText = `Soal ${currentQuestion + 1} dari ${quizData.length}`;
    questionTextEl.innerText = q.q;
    
    // Update progress bar
    const progress = ((currentQuestion + 1) / quizData.length) * 100;
    progressBarEl.style.width = `${progress}%`;

    optionsContainerEl.innerHTML = '';
    nextBtn.disabled = true;

    q.options.forEach((opt, index) => {
        const btn = document.createElement('button');
        btn.className = 'option-btn';
        btn.innerText = opt;
        btn.onclick = () => selectOption(index);
        optionsContainerEl.appendChild(btn);
    });
}

function selectOption(selectedIndex) {
    const q = quizData[currentQuestion];
    const buttons = optionsContainerEl.getElementsByClassName('option-btn');

    for (let i = 0; i < buttons.length; i++) {
        buttons[i].disabled = true;
        if (i === q.answer) {
            buttons[i].classList.add('correct');
        }
        if (i === selectedIndex && selectedIndex !== q.answer) {
            buttons[i].classList.add('incorrect');
        }
    }

    if (selectedIndex === q.answer) {
        score++;
        scoreCounterEl.innerText = `Skor: ${score}`;
    }

    nextBtn.disabled = false;
}

function nextQuestion() {
    currentQuestion++;
    if (currentQuestion < quizData.length) {
        loadQuestion();
    } else {
        showResults();
    }
}

function showResults() {
    document.getElementById('quiz-screen').style.display = 'none';
    const resultScreen = document.getElementById('result-screen');
    resultScreen.style.display = 'block';

    const finalScoreEl = document.getElementById('final-score');
    const feedbackTextEl = document.getElementById('feedback-text');
    
    finalScoreEl.innerText = `${score} / ${quizData.length}`;

    const percentage = (score / quizData.length) * 100;
    if (percentage >= 80) {
        feedbackTextEl.innerText = "Sangat Baik! Anda memahami konsep Psychological First Aid (PFA) dengan baik.";
    } else if (percentage >= 60) {
        feedbackTextEl.innerText = "Cukup Baik! Anda memiliki pemahaman dasar yang kuat, namun perlu memperdalam beberapa materi.";
    } else {
        feedbackTextEl.innerText = "Perlu Belajar Lagi. Pelajari kembali materi IFRC Psychosocial Centre Guide untuk memahami PFA secara komprehensif.";
    }
}

function restartQuiz() {
    currentQuestion = 0;
    score = 0;
    scoreCounterEl.innerText = `Skor: ${score}`;
    document.getElementById('quiz-screen').style.display = 'block';
    document.getElementById('result-screen').style.display = 'none';
    loadQuestion();
}
